# excavate-listen, built with whisper.cpp's own build (its backends as libraries loaded at run time: every processor
# variant, Vulkan, and CUDA's as an add-on). build.bat drives it.
cmake_minimum_required(VERSION 3.21)
project(excavate-listen C CXX)

set(WHISPER_SOURCE "" CACHE PATH "whisper.cpp's source, at the release Excavate uses")
if(NOT EXISTS "${WHISPER_SOURCE}/include/whisper.h")
    message(FATAL_ERROR "WHISPER_SOURCE must be whisper.cpp's source folder")
endif()
add_subdirectory("${WHISPER_SOURCE}" whisper.cpp)

add_executable(excavate-listen listen.cpp)
target_link_libraries(excavate-listen PRIVATE whisper parakeet ggml)
target_compile_features(excavate-listen PRIVATE cxx_std_17)
set_target_properties(excavate-listen PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

# Version details in every program and library of the package (Windows' file properties; virus scanners judge
# anonymous files harshly). The listener's build: also in its package's excavate-listen-build.txt and in
# indexer/src/models.rs (LISTEN_BUILD).
set(LISTEN_BUILD 3)
set(WHISPER_RELEASE "b5454")
set(LISTEN_VERSION "1.5454.${LISTEN_BUILD}.0")
file(WRITE "${CMAKE_BINARY_DIR}/bin/excavate-listen-build.txt" "${LISTEN_BUILD}\n")
string(REPLACE "." "," LISTEN_VERSION_COMMAS "${LISTEN_VERSION}")
function(listen_version target description copyright)
    if(NOT TARGET ${target})
        return()
    endif()
    get_target_property(kind ${target} TYPE)
    if(kind STREQUAL "EXECUTABLE")
        set(LISTEN_FILE_TYPE "VFT_APP")
        set(LISTEN_FILE "${target}.exe")
    elseif(kind STREQUAL "SHARED_LIBRARY" OR kind STREQUAL "MODULE_LIBRARY")
        set(LISTEN_FILE_TYPE "VFT_DLL")
        set(LISTEN_FILE "${target}.dll")
    else()
        return()
    endif()
    set(LISTEN_NAME "${target}")
    set(LISTEN_DESCRIPTION "${description}")
    set(LISTEN_COPYRIGHT "${copyright}")
    set(rc "${CMAKE_BINARY_DIR}/version/${target}.rc")
    set(res "${CMAKE_BINARY_DIR}/version/${target}.res")
    configure_file("${CMAKE_CURRENT_SOURCE_DIR}/version.rc.in" "${rc}" @ONLY)
    # Compiled on its own and handed to the linker: the targets' own compile options (ggml-vulkan's /MP) aren't the
    # resource compiler's.
    add_custom_command(OUTPUT "${res}" COMMAND "${CMAKE_RC_COMPILER}" /nologo /fo "${res}" "${rc}" DEPENDS "${rc}" VERBATIM)
    add_custom_target(${target}-version DEPENDS "${res}")
    add_dependencies(${target} ${target}-version)
    target_link_options(${target} PRIVATE "${res}")
endfunction()
set(WHISPER_COPYRIGHT "Copyright (c) 2023-2026 The ggml authors. MIT License.")
listen_version(excavate-listen "Excavate's listener: speech to text for Excavate's indexer" "Copyright (c) 2026 Amr Nader. MIT License.")
listen_version(whisper "whisper.cpp: Whisper speech recognition" "${WHISPER_COPYRIGHT}")
listen_version(parakeet "whisper.cpp: Parakeet speech recognition" "${WHISPER_COPYRIGHT}")
listen_version(ggml "ggml: tensor library" "${WHISPER_COPYRIGHT}")
listen_version(ggml-base "ggml: tensor library, base" "${WHISPER_COPYRIGHT}")
listen_version(ggml-vulkan "ggml: Vulkan backend (graphics cards)" "${WHISPER_COPYRIGHT}")
listen_version(ggml-cuda "ggml: CUDA backend (NVIDIA graphics cards)" "${WHISPER_COPYRIGHT}")
listen_version(ggml-cpu "ggml: processor backend" "${WHISPER_COPYRIGHT}")
foreach(variant x64 sse42 sandybridge ivybridge piledriver haswell skylakex cannonlake cascadelake icelake cooperlake zen4 alderlake sapphirerapids)
    listen_version(ggml-cpu-${variant} "ggml: processor backend (${variant})" "${WHISPER_COPYRIGHT}")
endforeach()
